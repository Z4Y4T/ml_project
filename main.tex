\chapterframe{1}{Постановка задачи машинного обучения}

\begin{frame}{Обучение с учителем: постановка задачи}
    \small
    \mlDefinition

    Пусть $(\mathsf X,Y)$ --- случайная пара с неизвестным совместным распределением
    $P_{\mathsf X,Y}$, где $\mathsf X\in\mathcal X$ --- объект, а $Y\in\mathcal Y$ --- его ответ.

    \mlDefinition

    \textit{Обучающей выборкой} называется набор реализаций
    \[
        D_n=\{(x_i,y_i)\}_{i=1}^{n},
        \qquad
        (\mathsf X_i,Y_i)\overset{\mathrm{iid}}{\sim}P_{\mathsf X,Y}.
    \]
    Здесь $(\mathsf X_i,Y_i)$ --- независимые одинаково распределённые случайные пары,
    а $(x_i,y_i)$ --- наблюдаемые значения.

    \mlTask

    По выборке $D_n$ построить функцию
    \[
        f:\mathcal X\to\widehat{\mathcal Y},
    \]
    для прогнозирования новых объектов из того же распределения.
    Здесь $\widehat{\mathcal Y}$ --- пространство прогнозов (в частности, вероятностей классов).

    \begin{center}
        \slideimg[width=0.82\linewidth,height=2.40cm,keepaspectratio]
        {img/supervised_learning_scheme.pdf}
    \end{center}
\end{frame}

\begin{frame}{Признаки и матричное представление выборки}
    \mlBody
    \mlDefinition

    Числовым \textit{признаковым описанием} $i$-го объекта называется вектор
    \[
        x_i=
        (x_{i1},\ldots,x_{id})^T\in\mathbb R^d,
    \]
    где $x_{ij}$ --- значение $j$-го признака у $i$-го объекта.

    Выборку удобно записывать в виде матрицы объектов--признаков
    \[
        X=
        \begin{pmatrix}
        x_1^T\\
        \vdots\\
        x_n^T
        \end{pmatrix}
        =
        \begin{pmatrix}
        x_{11}&\cdots&x_{1d}\\
        \vdots&\ddots&\vdots\\
        x_{n1}&\cdots&x_{nd}
        \end{pmatrix}
        \in\mathbb R^{n\times d},
        \qquad
        y=(y_1,\ldots,y_n)^T.
    \]

    \mlDefinition

    \begin{itemize}
        \item $\mathcal Y=\mathbb R$ --- \textit{regression} (регрессия);
        \item $\mathcal Y=\{1,\ldots,K\}$ --- \textit{classification} (классификация).
    \end{itemize}

    \mlExample\;
    Цена квартиры --- регрессия; определение класса изображения --- классификация.
\end{frame}

\begin{frame}{Модель и алгоритм обучения}
\mlBody
\mlDefinition\; \textit{Параметрическая модель} --- семейство функций
\[
 \mathcal F=\{f_\theta:\mathcal X\to\widehat{\mathcal Y}\mid\theta\in\Theta\},
\]
где $\Theta$ --- пространство параметров. При фиксированном $\theta$
получаем прогноз $\hat y=f_\theta(x)$.

\mlExample\; Прямые $f_{w,b}(x)=wx+b$ образуют семейство
с параметрами $\theta=(w,b)^T\in\mathbb R^2$.

\mlDefinition\; \textit{Алгоритм обучения} выбирает параметры по данным:
\[
 \boxed{A:(\mathcal X\times\mathcal Y)^n\to\Theta,\qquad
 \hat\theta=A(D_n)}.
\]
\mlConsequence\; Результат обучения --- конкретная функция
$f_{\hat\theta}$ из семейства $\mathcal F$.

\mlRemark\; Модель отвечает на вопрос «что можно построить?»,
алгоритм --- «как выбрать параметры по выборке?».
\end{frame}

\begin{frame}{Функция потерь: как измерить ошибку?}
\mlBody
\mlDefinition\; \textit{Функция потерь} задаёт стоимость ошибки прогноза:
\[
 \ell:\mathcal Y\times\widehat{\mathcal Y}\to\mathbb R_{\ge0},
 \qquad (y,\hat y)\mapsto\ell(y,\hat y).
\]
\mlExample\; В регрессии при числовом прогнозе $\hat y$:
\[
 \boxed{\ell(y,\hat y)=(y-\hat y)^2}.
\]
В классификации при прогнозировании \emph{класса} $\hat c\in\mathcal Y$:
\[
 \boxed{\ell(y,\hat c)=\mathbf1\{y\ne\hat c\}}.
\]
Здесь $\mathbf1\{A\}=1$, когда условие $A$ верно, и $0$ иначе.
Если модель выдаёт \emph{вероятность} класса, вместо $0/1$-потери
удобно использовать BCE (глава 2).

\mlRemark\; Потеря для обучения и метрика итогового качества
не обязаны совпадать: классификатор может обучаться по BCE,
а оцениваться по доле правильных ответов (accuracy).
\end{frame}

\begin{frame}{Истинный риск: ошибка на новых объектах}
\mlBody
Пусть случайные данные $(\mathsf X,Y)$ имеют неизвестное распределение
$P_{\mathsf X,Y}$.

\mlDefinition\; \textit{Истинный риск} --- математическое ожидание потери
на новом случайном объекте:
\[
 \boxed{R(\theta)=\mathbb E_{(\mathsf X,Y)\sim P_{\mathsf X,Y}}
 \bigl[\ell(Y,f_\theta(\mathsf X))\bigr]}.
\]
\mlRemark\; $\mathbb E$ означает усреднение по распределению
возможных новых объектов и их ответов, а не по одной наблюдаемой таблице.

\mlTask\; Идеально хотелось бы найти
$\hat\theta\in\arg\min_{\theta\in\Theta}R(\theta)$.
Но $P_{\mathsf X,Y}$ неизвестно, поэтому вычислить $R(\theta)$ точно обычно нельзя.

\mlConsequence\; Вместо истинного риска используют \emph{эмпирический}
--- усреднение потерь по имеющимся данным.
\end{frame}

\begin{frame}{Эмпирический риск: что известно по выборке?}
\mlBody
\mlDefinition\; Для $D_n=\{(x_i,y_i)\}_{i=1}^n$
\textit{эмпирический риск} равен
\[
 \boxed{\widehat R_n(\theta)=\frac1n\sum_{i=1}^{n}
 \ell\bigl(y_i,f_\theta(x_i)\bigr)}.
\]
Это вычисляемая функция параметров при фиксированной выборке.

\mlStatement\; При \emph{фиксированном} $\theta$ и конечном
математическом ожидании закон больших чисел даёт
\[
 \widehat R_n(\theta)\longrightarrow R(\theta),\qquad n\to\infty.
\]
\mlRemark\; Для параметров $\hat\theta$, выбранных
\emph{по этой же выборке}, такого вывода автоматически нет.
Поэтому качество проверяют на отдельных данных.

\mlIdea\; Подбор параметров превращается в задачу минимизации
конкретной функции $\widehat R_n$.
\end{frame}

\begin{frame}{Принцип ERM: обучение как оптимизация}
\mlBody
\mlDefinition\; \textit{Empirical Risk Minimization (ERM)} --- выбор
параметров с наименьшей средней потерей на обучающей выборке:
\[
 \boxed{\hat\theta\in\operatorname*{arg\,min}_{\theta\in\Theta}
 \widehat R_n(\theta)}.
\]
Здесь $\arg\min$ обозначает \emph{множество параметров, на которых
достигается минимум}, а не значение функции в минимуме.

\mlExample\; Для прямой $f_{w,b}(x)=wx+b$ нужно подобрать
два числа $w,b$, уменьшая среднюю квадратичную ошибку:
\[
 \widehat R_n(w,b)=\frac1n\sum_{i=1}^n(wx_i+b-y_i)^2.
\]
\mlConsequence\; Если $\Theta\subseteq\mathbb R^m$, то
$\widehat R_n:\Theta\to\mathbb R$ --- \emph{функция нескольких переменных}.

\mlRemark\; Минимальная ошибка на обучающих данных не означает
автоматически минимальную ошибку на новых объектах.
\end{frame}

\begin{frame}{Обобщающая способность и переобучение}
    \mlBody
    \mlDefinition

    \textit{Generalization} (обобщающая способность) --- способность модели,
    обученной на $D_{\mathrm{train}}$, сохранять качество на новых объектах
    из того же распределения.

    Практически сравнивают ошибку на обучающей и независимой выборках:
    \[
        \widehat R_{\mathrm{train}}(\hat\theta),
        \qquad
        \widehat R_{\mathrm{test}}(\hat\theta).
    \]

    \mlDefinition

    \textit{Overfitting} (переобучение) обычно проявляется как ситуация, когда
    \[
        \widehat R_{\mathrm{train}}(\hat\theta)
        \ll
        \widehat R_{\mathrm{test}}(\hat\theta).
    \]

    \textit{Underfitting} (недообучение) --- ситуация, когда
    обе ошибки остаются большими.

    \mlRemark

    Минимизация train loss --- средство, а не конечная цель:
    требуется малый истинный риск $R(\hat\theta)$.
    Разность $\widehat R_{\mathrm{test}}(\hat\theta)-
    \widehat R_{\mathrm{train}}(\hat\theta)$ называют \textit{разрывом обобщения};
    большой положительный разрыв сигнализирует о проблеме.
\end{frame}

\begin{frame}{Train, Validation и Test}
    \mlBody
    Для независимого контроля качества данные разделяют:
    \[
        D=
        D_{\mathrm{train}}
        \sqcup
        D_{\mathrm{validation}}
        \sqcup
        D_{\mathrm{test}}.
    \]

    \begin{itemize}
        \item $D_{\mathrm{train}}$ --- поиск параметров модели $\hat\theta$;
        \item $D_{\mathrm{validation}}$ --- выбор гиперпараметров (настроек, не входящих в $\theta$),
        архитектуры и варианта модели;
        \item $D_{\mathrm{test}}$ --- итоговая оценка качества после завершения выбора.
    \end{itemize}

    \mlRemark

    Test-выборка не должна использоваться при подборе модели:
    иначе она перестаёт быть независимой оценкой качества.

    \mlTask

    В следующих главах обозначим $L(\theta)=\widehat R_{\mathrm{train}}(\theta)$. Общий цикл:
    \[
        D_{\mathrm{train}}\to\mathcal F\to\ell
        \to\min L(\theta)\to\mathrm{validation}\to\mathrm{test}.
    \]

    \end{center}
\end{frame}

\chapterframe{2}{Линейные модели}
\begin{frame}{Linear Regression: определение модели}
    \mlBody
    \mlTask\; По $x\in\mathbb R^d$ предсказать число $y\in\mathbb R$.

    \mlDefinition\; \textit{Линейная регрессия} (Linear Regression):
    \[
        f_{w,b}(x)=w^Tx+b=\sum_{j=1}^{d}w_jx_j+b,
        \qquad w\in\mathbb R^d,\quad b\in\mathbb R.
    \]
    Здесь $w_j$ --- коэффициенты признаков, $b$ --- свободный член.

    \mlDefinition\; Для единой записи вводят \textit{фиктивный признак} $x_0=1$:
    \[
       \widetilde x=(1,x_1,\ldots,x_d)^T,\qquad
       \theta=(b,w_1,\ldots,w_d)^T,
       \qquad f_\theta(\widetilde x)=\widetilde x^T\theta.
    \]
    Итак, $\widetilde x,\theta\in\mathbb R^{d+1}$. Для выборки обозначим через $\widetilde X$ матрицу строк $\widetilde x_i^T$.

    \mlExample\; При $d=1$ получаем прямую
    $\hat y=wx+b$; например, $w=2$, $b=1$ дают
    $f(3)=2\cdot3+1=7$.

    \mlRemark\; Линейность здесь означает линейность \textit{по параметрам}:
    $\theta_0+\theta_1x+\theta_2x^2$ также линейна по $\theta$.
\end{frame}

\begin{frame}{Геометрия линейной регрессии: данные и остатки}
    \mlBody
    Каждая точка $(x_i,y_i)$ --- наблюдение; линия --- прогноз
    $\hat y_i=wx_i+b$. При изменении $w,b$ меняются прогнозы и ошибки.

    \begin{center}
        \slideimg[width=0.93\linewidth,height=5.36cm,keepaspectratio]
        {img/linear_regression.pdf}
    \end{center}
    \mlRemark\; Вертикальное отклонение точки от прямой --- остаток
    $r_i=y_i-\hat y_i$; минимизировать будем \textit{сумму квадратов} таких отклонений.
\end{frame}

\begin{frame}{Остатки и Mean Squared Error}
    \mlBody
    \mlDefinition\; \textit{Остаток} (residual) на $i$-м объекте:
    \[
        r_i=y_i-\hat y_i=y_i-f_\theta(\widetilde x_i).
    \]
    \mlDefinition\; Квадратичная потеря и её среднее по выборке:
    \[
        \ell(y_i,\hat y_i)=(y_i-\hat y_i)^2,
        \qquad
        \boxed{L(\theta)=\operatorname{MSE}(\theta)
        =\frac1n\sum_{i=1}^{n}r_i^2}.
    \]
    \mlExample\; Пусть $x=(0,1,2)$, $y=(1,2,2)$, а $f(x)=x+1$.
    Тогда
    \[
        \hat y=(1,2,3),\quad r=(0,0,-1),\quad
        L=\frac{0^2+0^2+(-1)^2}{3}=\frac13.
    \]
    \mlRemark\; Возведение в квадрат убирает знак ошибок и сильнее
    штрафует большие отклонения: $|r|=2$ даёт потерю $4$, а $|r|=1$ --- $1$.

    \mlTask\; Найти $\hat\theta\in\operatorname*{arg\,min}_{\theta}L(\theta)$.
\end{frame}

\begin{frame}{Матричная запись MSE}
    \mlBody
    Расширенная матрица $\widetilde X$ содержит единичный первый столбец и $d$ исходных признаков:
    \[
        \widetilde X=\begin{pmatrix}
           1 & x_{11}&\cdots&x_{1d}\\
           \vdots&\vdots&&\vdots\\
           1&x_{n1}&\cdots&x_{nd}
        \end{pmatrix}\in\mathbb R^{n\times(d+1)},
        \quad \theta\in\mathbb R^{d+1}.
    \]
    Поэтому вектор всех предсказаний и остатков имеет вид
    \[
        \hat y=\widetilde X\theta\in\mathbb R^n,\qquad
        r=y-\widetilde X\theta\in\mathbb R^n.
    \]
    \mlDefinition\; \textit{Евклидова норма} вектора
    $r=(r_1,\ldots,r_n)^T$ --- его длина:
    $\|r\|_2=\sqrt{r_1^2+\cdots+r_n^2}$.
    Поэтому $\|r\|_2^2=r^Tr=\sum_i r_i^2$ и
    \[
        \boxed{L(\theta)=\frac1n\|y-\widetilde X\theta\|_2^2
        =\frac1n(y-\widetilde X\theta)^T(y-\widetilde X\theta)}.
    \]
    \mlConsequence\; Это функция от $(d+1)$ переменной, а не от объектов
    выборки: при обучении меняется $\theta$, тогда как $X$ и $y$ зафиксированы.
\end{frame}

\begin{frame}{MSE как функция нескольких переменных}
    \mlBody
    Для единственного признака $\theta=(b,w)$ и
    \[
        L(w,b)=\frac1n\sum_{i=1}^{n}(y_i-wx_i-b)^2,
        \qquad L:\mathbb R^2\to\mathbb R.
    \]
    При каждом $(w,b)$ выбирается своя прямая и вычисляется её ошибка.

    \mlStatement\; $L(w,b)$ --- квадратичная функция по двум параметрам;
    её график --- поверхность в пространстве $(w,b,L)$.

    \mlRemark\; Точки минимума соответствуют параметрам прямой,
    наилучшей по MSE. Как найти их численно, разберём в главе 3.

    \begin{center}
        \slideimg[width=0.90\linewidth,height=3.66cm,keepaspectratio]
        {img/mse_surface.pdf}
    \end{center}
\end{frame}

\begin{frame}{Линии уровня MSE: геометрический смысл}
\mlBody
\mlDefinition\; \textit{Линия уровня} функции $L(w,b)$ при $c\in\mathbb R$:
\[
    \mathcal C_c=\{(w,b):L(w,b)=c\}.
\]
Чем ближе линия уровня к минимуму, тем меньшей сумме квадратов
остатков соответствуют параметры.
\begin{center}
    \slideimg[width=0.91\linewidth,height=4.15cm,keepaspectratio]{img/mse_contours.pdf}
\end{center}
\mlRemark\; В точках, где $\nabla L\ne0$, градиент перпендикулярен
линии уровня: при движении вдоль неё значение $L$ локально не меняется.
\end{frame}


\begin{frame}{Как дифференцировать MSE: случай $d=1$}
    \mlBody
    Для $f_{w,b}(x)=wx+b$ положим $e_i=wx_i+b-y_i$.
    Тогда $L(w,b)=\frac1n\sum_i e_i^2$.

    По \textit{правилу цепочки}, поскольку
    $\partial e_i/\partial w=x_i$, а $\partial e_i/\partial b=1$,
    \[
       \frac{\partial e_i^2}{\partial w}=2e_ix_i,
       \qquad
       \frac{\partial e_i^2}{\partial b}=2e_i.
    \]
    \mlStatement\; Частные производные функции потерь:
    \[
        \boxed{\frac{\partial L}{\partial w}
        =\frac2n\sum_{i=1}^{n}(wx_i+b-y_i)x_i},
    \]
    \[
        \boxed{\frac{\partial L}{\partial b}
        =\frac2n\sum_{i=1}^{n}(wx_i+b-y_i)}.
    \]
    \mlRemark\; Градиент $\nabla L=(\partial L/\partial b,
    \partial L/\partial w)^T$ показывает, как ошибка меняется
    при малом изменении каждого параметра.
\end{frame}

\begin{frame}{Градиент MSE в общем случае}
    \mlBody
    Запишем $e_i=\widetilde x_i^T\theta-y_i$; тогда
    \[
       L(\theta)=\frac1n\sum_{i=1}^{n}e_i^2,
       \qquad \frac{\partial e_i}{\partial\theta_j}=\widetilde x_{ij},
       \quad j=0,\ldots,d.
    \]
    Снова применяем правило цепочки:
    \[
       \frac{\partial L}{\partial\theta_j}
       =\frac1n\sum_i 2e_i\frac{\partial e_i}{\partial\theta_j}
       =\frac2n\sum_i \widetilde x_{ij}(\widetilde x_i^T\theta-y_i).
    \]
    Здесь $\widetilde X^T(\widetilde X\theta-y)$ --- вектор, $j$-я компонента которого
    равна сумме в правой части без множителя $2/n$.

    \mlStatement\; Собирая частные производные, получаем
    \[
        \boxed{\nabla_\theta L(\theta)=\frac2n\widetilde X^T(\widetilde X\theta-y)}
        \quad\in\mathbb R^{d+1}.
    \]
    \mlConsequence\; Минимум гладкой функции (во внутренней точке)
    может достигаться только при $\nabla_\theta L=0$.
\end{frame}

\begin{frame}{Logistic Regression: зачем нужна сигмоида}
    \mlBody
    \mlTask\; Предсказать класс $Y\in\{0,1\}$ и вероятность
    $p_\theta(x)=P_\theta(Y=1\mid \mathsf X=x)$.

    \mlDefinition\; Сначала вычисляется \textit{линейный score}
    \[
        z=w^Tx+b=\widetilde x^T\theta\in\mathbb R.
    \]
    Но $z$ не обязан принадлежать $[0,1]$, поэтому сам по себе
    не может быть вероятностью.

    \mlDefinition\; \textit{Odds} (шансы) события при вероятности $p$
    равны $p/(1-p)$. \textit{Log-odds} --- их логарифм:
    \[
       \operatorname{logit}(p)=\log\frac{p}{1-p}.
    \]
    \mlStatement\; В \textit{логистической регрессии} принимают модель
    \[
       \boxed{\operatorname{logit}(p_\theta(x))=w^Tx+b=z}.
    \]
    \mlExample\; При $p=3/4$ шансы равны $3$, а log-odds $=\log3$.
\end{frame}

\begin{frame}{Вывод сигмоиды из log-odds}
    \mlBody
    Предположим, что $\log\frac{p}{1-p}=z$. Выразим $p$:
    \[
       \log\frac{p}{1-p}=z
       \quad\Longrightarrow\quad
       \frac{p}{1-p}=e^z,
    \]
    \[
       p=e^z(1-p)
       \quad\Longrightarrow\quad
       p(1+e^z)=e^z,
    \]
    \[
       \boxed{p=\frac{e^z}{1+e^z}=\frac1{1+e^{-z}}=\sigma(z)}.
    \]
    \mlDefinition\; $\sigma:\mathbb R\to(0,1)$ ---
    \textit{sigmoid} (логистическая функция).

    \mlStatement\; $\sigma(0)=1/2$,
    $\lim_{z\to-\infty}\sigma(z)=0$,
    $\lim_{z\to+\infty}\sigma(z)=1$.

    По правилу цепочки
    \[
       \boxed{\sigma'(z)=\frac{e^{-z}}{(1+e^{-z})^2}
       =\sigma(z)\bigl(1-\sigma(z)\bigr)>0}.
    \]
    Поэтому вероятность класса $1$ растёт вместе со score $z$.
\end{frame}

\begin{frame}{График сигмоиды}
    \mlBody
    \mlStatement\; Линейный score $z$ переводится в вероятность
    $p=\sigma(z)$; центральная точка графика имеет координаты $(0,1/2)$.
    \begin{center}
        \slideimg[width=0.93\linewidth,height=5.25cm,keepaspectratio]
        {img/sigmoid.pdf}
    \end{center}
    \mlExample\; Если $z=\log3$, то $p=3/4$;
    если $z=-\log3$, то $p=1/4$.
\end{frame}



\begin{frame}{Геометрия логистической классификации}
    \mlBody
    При пороге $1/2$ и монотонности $\sigma$:
    \[
       \widehat y=1\iff p_\theta(x)\ge\tfrac12
       \iff w^Tx+b\ge0.
    \]
    \mlDefinition\; Граница решения: $\{x:w^Tx+b=0\}$ --- гиперплоскость (если $w\ne0$).
    \begin{center}
        \slideimg[width=0.93\linewidth,height=3.82cm,keepaspectratio]
        {img/logistic_boundary.pdf}
    \end{center}
    \mlRemark\; На графике показана линейная граница,
    хотя сама вероятность $p_\theta(x)=\sigma(w^Tx+b)$ нелинейна по $x$.
    Например, для $w=(1,-1)^T$, $b=0$ граница имеет вид $x_1=x_2$.
\end{frame}

\begin{frame}{Модель Бернулли и функция правдоподобия}
    \mlBody
    Для объекта $x_i$ обозначим
    $p_i=\sigma(\widetilde x_i^T\theta)$.

    \mlDefinition\; При фиксированном $x_i$ модель задаёт
    \[
       Y_i\mid \mathsf X_i=x_i\sim\operatorname{Bern}(p_i),
       \qquad
       P_\theta(Y_i=y_i\mid x_i)=p_i^{y_i}(1-p_i)^{1-y_i}.
    \]
    Действительно, если $y_i=1$, получаем $p_i$;
    если $y_i=0$, получаем $1-p_i$.

    При условной независимости ответов \textit{likelihood}
    (правдоподобие) всей наблюдаемой выборки равно
    \[
       \boxed{\mathcal L(\theta)=
       \prod_{i=1}^{n}p_i^{y_i}(1-p_i)^{1-y_i}}.
    \]
    \mlTask\; \textit{Maximum Likelihood Estimation} (MLE):
    \[
        \hat\theta\in\operatorname*{arg\,max}_\theta\mathcal L(\theta).
    \]
    \mlRemark\; $\mathcal L(\theta)$ рассматривается как функция параметров:
    ответы $y_i$ уже известны, изменяются только $p_i$.
\end{frame}

\begin{frame}{Вывод Binary Cross-Entropy из MLE}
    \mlBody
    Логарифм строго возрастает, поэтому максимумы $\mathcal L$
    и $\log\mathcal L$ достигаются при тех же параметрах.

    Применяем $\log(ab)=\log a+\log b$ и $\log(a^c)=c\log a$:
    \[
       \log\mathcal L(\theta)
       =\sum_{i=1}^{n}\Bigl(y_i\log p_i
       +(1-y_i)\log(1-p_i)\Bigr).
    \]
    Чтобы получить \textit{задачу минимизации}, меняем знак и делим
    на $n$ (положительный множитель не меняет оптимум):
    \[
       \boxed{L(\theta)=-\frac1n\sum_{i=1}^{n}
       \Bigl[y_i\log p_i+(1-y_i)\log(1-p_i)\Bigr]}.
    \]
    \mlDefinition\; $L$ --- \textit{Binary Cross-Entropy} (BCE),
    или средняя отрицательная логарифмическая вероятность правильного ответа.

    \mlExample\; Для одного наблюдения с $y=1$ потеря равна $-\log p$:
    прогноз $p=0.9$ даёт примерно $0.105$, а $p=0.1$ --- примерно $2.303$.
    Уверенная ошибка наказывается сильнее.

    \mlConsequence\; $\operatorname*{arg\,max}_\theta\mathcal L(\theta)
    =\operatorname*{arg\,min}_\theta L(\theta)$.
\end{frame}

\begin{frame}{Градиент BCE: правило цепочки}
    \mlBody
    Для одного объекта положим $z_i=\widetilde x_i^T\theta$,
    $p_i=\sigma(z_i)$ и
    \[
       \ell_i=-y_i\log p_i-(1-y_i)\log(1-p_i).
    \]
    Последовательно дифференцируем:
    \[
       \frac{\partial\ell_i}{\partial p_i}
       =-\frac{y_i}{p_i}+\frac{1-y_i}{1-p_i}
       =\frac{p_i-y_i}{p_i(1-p_i)},
    \]
    \[
       \frac{\partial p_i}{\partial z_i}=p_i(1-p_i),
       \qquad
       \boxed{\frac{\partial\ell_i}{\partial z_i}=p_i-y_i}.
    \]
    Ещё один шаг по правилу цепочки:
    \[
       \frac{\partial z_i}{\partial\theta_j}=\widetilde x_{ij}
       \quad\Longrightarrow\quad
       \frac{\partial\ell_i}{\partial\theta_j}
       =(p_i-y_i)\widetilde x_{ij}.
    \]
    \mlStatement\; Для всей выборки
    \[
        \boxed{\nabla_\theta L(\theta)=\frac1n\widetilde X^T(p-y)},
        \qquad p=(p_1,\ldots,p_n)^T.
    \]
\end{frame}

\chapterframe{3}{Градиентные методы оптимизации}
\begin{frame}{Зачем нужна численная оптимизация}
    \mlBody
    В предыдущей главе обучение свелось к поиску минимума
    эмпирического риска:
    \[
        \boxed{\hat\theta\in\operatorname*{arg\,min}_{\theta\in\Theta}L(\theta)},
        \qquad L:\mathbb R^m\to\mathbb R.
    \]

    \mlExample\; Для рассмотренных линейных моделей
    \[
    \begin{aligned}
        L_{\mathrm{MSE}}(\theta)&=\frac1n\|\widetilde X\theta-y\|_2^2,\\
        L_{\mathrm{BCE}}(\theta)&=-\frac1n\sum_{i=1}^n
        \bigl[y_i\log p_i+(1-y_i)\log(1-p_i)\bigr].
    \end{aligned}
    \]
    Для MSE функция потерь квадратичная; для BCE общего явного решения нет.

    \mlTask\; Построить последовательность $\theta_0,\theta_1,\ldots$,
    вдоль которой значение функции потерь становится меньше.

    \mlIdea\; Если известны частные производные $L$, можно использовать
    локальную информацию о её изменении, не решая уравнение минимума явно.
    Следующий шаг --- выбрать направление, в котором линейное
    приближение $L$ убывает быстрее всего.
\end{frame}



\begin{frame}{Скалярное произведение и длина вектора}
\mlBody
Перед геометрией градиента уточним обозначения для
$u,v\in\mathbb R^m$.

\mlDefinition\; \textit{Скалярное произведение} --- число
\[
    \boxed{\langle u,v\rangle=u^Tv=\sum_{j=1}^m u_jv_j}.
\]
\mlDefinition\; \textit{Евклидова норма} --- длина вектора:
\[
    \boxed{\|v\|_2=\sqrt{\langle v,v\rangle}
    =\sqrt{v_1^2+\cdots+v_m^2}}.
\]
\mlExample\; Для $v=(3,4)^T$ получаем $\|v\|_2=5$;
единичный вектор в этом направлении равен $v/\|v\|_2=(3/5,4/5)^T$.

\mlRemark\; \textit{Единичным} называется вектор с нормой $1$.
Неравенство Коши--Буняковского:
$|\langle u,v\rangle|\le\|u\|_2\|v\|_2$.
\end{frame}

\begin{frame}{Почему именно антиградиент?}
    \mlBody
    Пусть $L:\mathbb R^m\to\mathbb R$ дифференцируема, а $\nabla L$ --- вектор её частных производных.\par\medskip
    \mlDefinition\; \textit{Производная по направлению} единичного
    вектора $v$, $\|v\|_2=1$, задаётся равенством
    \[
        D_vL(\theta)=\langle\nabla L(\theta),v\rangle.
    \]
    По неравенству Коши--Буняковского
    \[
        -\|\nabla L(\theta)\|_2\le D_vL(\theta)
        \le\|\nabla L(\theta)\|_2.
    \]
    \mlStatement\; При $\nabla L(\theta)\ne0$ наибольшее
    \emph{локальное} убывание среди единичных направлений достигается при
    \[
        \boxed{v_{\min}=-\frac{\nabla L(\theta)}
        {\|\nabla L(\theta)\|_2}}.
    \]
    \mlRemark\; При $\Delta\theta=-\eta\nabla L(\theta)$
    линейное приближение даёт изменение
    $-\eta\|\nabla L(\theta)\|_2^2<0$ при $\nabla L\ne0$.
    Реальный конечный шаг может вести себя иначе: вывод локальный.
\end{frame}

\begin{frame}{Геометрия градиента и линий уровня}
    \mlBody
    На карте линий уровня $L(\theta)=c$ градиент в регулярной точке
    перпендикулярен линии уровня и направлен в сторону возрастания $L$.
    Антиградиент показывает направление наиболее быстрого
    \emph{локального} убывания.

\begin{frame}{Gradient Descent: формула и алгоритм}
    \mlBody
    \mlDefinition\; \textit{Gradient Descent (градиентный спуск)} ---
    итерационный метод, в котором
    \[
        \boxed{\theta_{t+1}=\theta_t-\eta_t\nabla L(\theta_t)},
        \qquad \eta_t>0.
    \]
    Здесь $\theta_t\in\mathbb R^m$ --- текущие параметры,
    $\eta_t$ --- \textit{learning rate} (размер шага обучения).

    \mlAlgorithm
    \begin{enumerate}
        \item Задать начальное приближение $\theta_0$ и правило выбора $\eta_t$.
        \item Вычислить $g_t=\nabla L(\theta_t)$.
        \item Обновить параметры: $\theta_{t+1}=\theta_t-\eta_t g_t$.
        \item Повторять до выполнения критерия остановки.
    \end{enumerate}

    \mlRemark\; Возможные критерии: малость $\|g_t\|$,
    изменение $L(\theta_t)$ или ограничение числа итераций.
    Ненулевой шаг сам по себе не гарантирует уменьшения функции.
\end{frame}

\begin{frame}{Пример: несколько итераций вручную}
    \mlBody
    \mlExample\; Минимизируем $L(u)=(u-2)^2$ из точки $u_0=0$
    при постоянном шаге $\eta=\frac14$.
    \[
        L'(u)=2(u-2),\qquad
        u_{t+1}=u_t-\tfrac14\,2(u_t-2)
        =\tfrac12u_t+1.
    \]
    \begin{center}
        \begin{tabular}{c|c|c|c}
            $t$ & $u_t$ & $L'(u_t)$ & $L(u_t)$\\\hline
            $0$ & $0$ & $-4$ & $4$\\
            $1$ & $1$ & $-2$ & $1$\\
            $2$ & $\frac32$ & $-1$ & $\frac14$\\
            $3$ & $\frac74$ & $-\frac12$ & $\frac1{16}$
        \end{tabular}
    \end{center}

    \mlConsequence\; Каждый шаг сокращает расстояние до $u^*=2$
    вдвое; ошибка по функции уменьшается вчетверо.
    \[
        u_t=2-2\left(\tfrac12\right)^t,
        \qquad L(u_t)=4\left(\tfrac14\right)^t\to0.
    \]
\end{frame}

\begin{frame}{Полный и стохастический градиенты}
    \mlBody
    Эмпирический риск является средним по выборке:
    \[
        L(\theta)=\frac1n\sum_{i=1}^n\ell_i(\theta),
        \qquad \ell_i(\theta)=\ell(y_i,f_\theta(x_i)).
    \]
    \mlDefinition\; При \textit{Batch Gradient Descent}
    используется градиент всей суммы:
    \[
        g_t=\nabla L(\theta_t)
        =\frac1n\sum_{i=1}^n\nabla\ell_i(\theta_t).
    \]
    \mlDefinition\; В \textit{Mini-batch SGD} вычисляют
    приближение по случайной подвыборке индексов $B_t$:
    \[
        \boxed{g_t=\frac1{|B_t|}\sum_{i\in B_t}
        \nabla\ell_i(\theta_t)},\qquad
        \theta_{t+1}=\theta_t-\eta_t g_t.
    \]
    При равномерном выборе индексов такое среднее в среднем
    направлено как полный градиент.

    \mlRemark\; Mini-batch дешевле одной полной итерации,
    но даёт шумное направление. Далее $g_t$ --- вычисленный на шаге градиент
    (полный или по mini-batch).
\end{frame}

\begin{frame}{Momentum: зачем учитывать предыдущие шаги?}
    \mlBody
    Вытянутые линии уровня означают разные масштабы изменения функции
    по разным направлениям. Обычный GD может сильно колебаться
    поперёк узкой долины и медленно идти вдоль неё.

    \mlIdea\; Сохранять накопленное направление движения,
    сглаживая изменения градиента между итерациями.

    \mlDefinition\; \textit{Momentum (метод момента)}:
    \[
    \begin{aligned}
        v_{t+1}&=\beta v_t+g_t,\qquad v_0=0,\\
        \theta_{t+1}&=\theta_t-\eta v_{t+1},
        \qquad 0\le\beta<1.
    \end{aligned}
    \]
    Здесь $v_t$ --- накопленное направление,
    $\beta$ --- коэффициент сохранения предыдущего направления.

    \mlRemark\; Если $\beta=0$, получаем обычный GD.
    При близких направлениях градиентов они накапливаются;
    при чередовании знаков в одной координате --- частично компенсируются.
\end{frame}

\begin{frame}{Momentum: геометрия узкой долины}
\mlBody
\mlExample\; Для $L(u,v)=10u^2+v^2$ имеем
\[
   \nabla L(u,v)=(20u,2v)^T.
\]
Изменение вдоль координаты $u$ намного резче; обычный GD с заметным
шагом может многократно пересекать долину поперёк.
\begin{center}
    \slideimg[width=0.88\linewidth,height=3.95cm,keepaspectratio]{img/momentum_valley.pdf}
\end{center}
\mlConsequence\; Momentum накапливает устойчивое направление вдоль
долины и частично гасит чередующиеся поперечные шаги.
\end{frame}


\begin{frame}{Nesterov Momentum: шаг с опережением}
    \mlBody
    \mlIdea\; Обычный Momentum вычисляет градиент в $\theta_t$,
    а затем делает инерционный шаг. Метод Нестерова сначала учитывает
    предполагаемое перемещение по инерции.

    \mlDefinition\; \textit{Nesterov Momentum (метод Нестерова)}
    в одной из стандартных форм записи:
    \[
    \begin{aligned}
        \theta_t^{\mathrm{look}}&=\theta_t-\eta\beta v_t,\\
        g_t&=\nabla L(\theta_t^{\mathrm{look}}),\\
        v_{t+1}&=\beta v_t+g_t,\\
        \theta_{t+1}&=\theta_t-\eta v_{t+1}.
    \end{aligned}
    \]
    Для Momentum градиент берётся в $\theta_t$,
    для Nesterov --- в $\theta_t^{\mathrm{look}}$.

    \mlRemark\; При $\beta=0$ обе формулы переходят в обычный GD.
    Термин «опережение» описывает \emph{место вычисления градиента},
    а не предсказание точного положения минимума.
\end{frame}

\begin{frame}{AdaGrad: накопление квадратов градиентов}
    \mlBody
    \mlDefinition\; \textit{AdaGrad (адаптивный градиентный метод)}
    хранит сумму квадратов предыдущих градиентов.
    Здесь $u\odot v=(u_1v_1,\ldots,u_mv_m)^T$ ---
    \textit{покоординатное произведение}:
    \[
    \begin{aligned}
        G_{t+1}&=G_t+g_t\odot g_t,\qquad G_0=0,\\
        \theta_{t+1}&=\theta_t-\eta
        \frac{g_t}{\sqrt{G_{t+1}}+\varepsilon}.
    \end{aligned}
    \]
    Квадратный корень, деление и сложение с $\varepsilon>0$
    выполняются по координатам.

    \mlExample\; Если первый градиент $g_0=(2,1)^T$,
    то $G_1=(4,1)^T$. При малом $\varepsilon$ масштабы шага
    равны приблизительно $\eta/2$ и $\eta$.

    \mlConsequence\; Для координаты $j$ эффективный шаг имеет масштаб
    \[
       \eta_{t,j}=\frac{\eta}{\sqrt{G_{t+1,j}}+\varepsilon}.
    \]
    Поскольку $G_{t,j}$ не убывает, этот масштаб может со временем
    стать очень малым --- главный недостаток AdaGrad.
\end{frame}

\begin{frame}{RMSProp: не суммировать всю историю}
    \mlBody
    \mlIdea\; В AdaGrad квадрат градиента накапливается навсегда.
    Заменим сумму \emph{экспоненциальным скользящим средним},
    в котором давние градиенты постепенно забываются.

    \mlDefinition\; \textit{RMSProp}:
    \[
    \begin{aligned}
        s_{t+1}&=\rho s_t+(1-\rho)(g_t\odot g_t),
        \qquad s_0=0,\\
        \theta_{t+1}&=\theta_t-
        \eta\frac{g_t}{\sqrt{s_{t+1}}+\varepsilon},
        \qquad 0\le\rho<1.
    \end{aligned}
    \]
    \mlRemark\; При постоянном масштабе градиентов $s_t$
    стабилизируется, а не растёт без ограничения,
    как сумма в AdaGrad. Поэтому масштаб шага может сохраняться и на поздних итерациях.
\end{frame}


\begin{frame}{Adam: инерция и адаптивный масштаб}
\mlBody
\mlIdea\; Momentum сглаживает направление градиента, RMSProp --- его покоординатный масштаб. \textit{Adam} объединяет эти идеи.

При $g_t=\nabla L(\theta_t)$, $m_0=s_0=0$:
\[
 \begin{aligned}
 m_{t+1}&=\beta_1m_t+(1-\beta_1)g_t,\\
 s_{t+1}&=\beta_2s_t+(1-\beta_2)(g_t\odot g_t).
 \end{aligned}
\]
Поправки на нулевую инициализацию:
\[
 \widehat m_{t+1}=\frac{m_{t+1}}{1-\beta_1^{t+1}},\qquad
 \widehat s_{t+1}=\frac{s_{t+1}}{1-\beta_2^{t+1}}.
\]
\mlDefinition\; Шаг Adam ($\varepsilon>0$):
\[
 \boxed{\theta_{t+1}=\theta_t-\eta\frac{\widehat m_{t+1}}{\sqrt{\widehat s_{t+1}}+\varepsilon}}.
\]
\mlRemark\; Квадраты, корни, деление и прибавление $\varepsilon$
в формулах векторов выполняются покоординатно;
$\odot$ --- покоординатное произведение.
Adam не гарантирует превосходства на любой задаче.
\end{frame}

    \mlBody
    На одной функции и из одной стартовой точки методы
    могут выбирать разные направления и длины шагов.

\chapterframe{4}{Решающие деревья и ансамбли}

\begin{frame}{Что меняется при переходе от линейной модели к дереву?}
\mlBody
\mlTask\; Построить правило, которое умеет выбирать разные ответы в разных областях пространства признаков.

Линейная модель использует одну формулу $f_\theta(x)$ для всей области. Дерево вместо этого последовательно проверяет условия на признаки:
\[
  x_j\le t\quad\text{или}\quad x_j>t.
\]
\mlDefinition\; \textit{Решающее дерево} (Decision Tree) --- модель из внутренних вершин с условиями и листьев с прогнозами.

\mlExample\; Для стоимости жилья: «площадь $>50$?» --- затем «район центральный?» --- в достигнутом листе прогноз среднего значения цены.

\mlRemark\; Сами признаки, пороги и структура дерева \emph{выбираются} по обучающей выборке. Это дискретная задача, не похожая на настройку коэффициентов прямой.
\end{frame}

\begin{frame}{Один split: формальное разбиение выборки}
\mlBody
Пусть $S\subseteq D_{\mathrm{train}}$ --- объекты, попавшие в текущую вершину дерева.

\mlDefinition\; Пара $(j,t)$ определяет два непересекающихся множества:
\[
 S_L(j,t)=\{(x_i,y_i)\in S:x_{ij}\le t\},
\]
\[
 S_R(j,t)=\{(x_i,y_i)\in S:x_{ij}>t\}.
\]
Вместе они образуют $S=S_L\sqcup S_R$.

\mlRemark\; Для числовых признаков достаточно проверять пороги между соседними различными отсортированными значениями признака.

\begin{center}
\slideimg[width=0.89\linewidth,height=3.62cm,keepaspectratio]{img/tree_split.pdf}
\end{center}
\end{frame}

\begin{frame}{Регрессия: оцениваем разброс ответов через MSE}
\mlBody
\mlTask\; Если ответы $y_i$ --- вещественные числа (например, цены квартир),
дереву нужно разбивать объекты на группы с близкими значениями $y_i$.

\mlDefinition\; Для непустой вершины $S$ обозначим средний ответ
\[
  \bar y_S=\frac1{|S|}\sum_{(x_i,y_i)\in S} y_i.
\]
Её \textit{неоднородностью} можно считать среднюю квадратичную ошибку
относительно этого среднего:
\[
  \boxed{H_{\mathrm{reg}}(S)=\frac1{|S|}
  \sum_{(x_i,y_i)\in S}(y_i-\bar y_S)^2.}
\]

\mlIdea\; Эта формула --- знакомая MSE из главы 2. Только здесь
предсказание внутри вершины постоянно и равно $\bar y_S$.

\mlExample\; Для ответов $1,1,5,5$ имеем $\bar y_S=3$ и
\[
 H_{\mathrm{reg}}(S)=\frac{(1-3)^2+(1-3)^2+(5-3)^2+(5-3)^2}{4}=4.
\]
Если разделить объекты на группы $(1,1)$ и $(5,5)$, то MSE
в каждой из двух новых вершин равна нулю.
\end{frame}

\begin{frame}{Классификация: индекс Джини}
\mlBody
\mlTask\; Для классификации ответы --- номера классов, поэтому
обычная MSE между их номерами зависит от произвольной нумерации.
Вместо неё оцениваем \emph{смешанность классов} в вершине.

\mlDefinition\; Пусть $p_k(S)$ --- доля объектов класса $k$ среди
объектов вершины $S$, причём $\sum_{k=1}^{K}p_k(S)=1$.
\textit{Индекс Джини} задаётся формулой
\[
  \boxed{H_{\mathrm{Gini}}(S)=1-\sum_{k=1}^{K}p_k(S)^2.}
\]

\mlIdea\; Это вероятность того, что два \emph{независимых}
случайных выбора класса по распределению $(p_1,\ldots,p_K)$
дадут разные классы.

\mlExample\; Для двух классов с долями $p$ и $1-p$:
\[
 H_{\mathrm{Gini}}=1-p^2-(1-p)^2=2p(1-p).
\]
При $p=1/2$ получаем $H_{\mathrm{Gini}}=1/2$, а при $p=0$ или
$p=1$ неоднородность равна нулю: все объекты одного класса.
\end{frame}

\begin{frame}{Классификация: энтропия Шеннона}
\mlBody
\mlDefinition\; Ещё одна мера неоднородности ---
\textit{энтропия Шеннона}:
\[
  \boxed{H_{\mathrm{ent}}(S)=-\sum_{k=1}^{K}p_k(S)\ln p_k(S).}
\]
Здесь $\ln$ --- натуральный логарифм, а $0\ln 0$ по определению
считается равным нулю.

\mlIdea\; Величина $-\ln p_k$ показывает, насколько неожиданно
появление класса $k$. Энтропия --- среднее значение этой
неопределённости по всем классам.

\mlExample\; Для двух классов:
\[
  H_{\mathrm{ent}}(1/2,1/2)=\ln 2,
  \qquad H_{\mathrm{ent}}(1,0)=0.
\]
\mlRemark\; И Джини, и энтропия равны нулю в полностью однородной
вершине и максимальны, когда классы представлены поровну.
Они измеряют одну идею, но численно различаются и могут выбирать
разные разбиения.
\end{frame}

\begin{frame}{Общий критерий качества разбиения}
\mlBody
\mlDefinition\; Теперь можно обозначить выбранную меру
неоднородности одной буквой $H(S)$:
\[
 H(S)=\begin{cases}
 H_{\mathrm{reg}}(S), & \text{регрессия (MSE)},\\
 H_{\mathrm{Gini}}(S)\ \text{или}\ H_{\mathrm{ent}}(S), & \text{классификация}.
 \end{cases}
\]

Пусть проверка $x_j\le t$ разделила вершину $S$ на
непустые $S_L=S_L(j,t)$ и $S_R=S_R(j,t)$.
\textit{Уменьшение неоднородности} при этом разбиении:
\[
 \boxed{Q(S,j,t)=H(S)
 -\frac{|S_L|}{|S|}H(S_L)
 -\frac{|S_R|}{|S|}H(S_R).}
\]

\mlIdea\; Из неоднородности \emph{до} разбиения вычитается
средняя неоднородность \emph{после}, причём вес каждой группы ---
доля объектов в ней: $|S_L|/|S|$ и $|S_R|/|S|$.

\mlConsequence\; Хорошее разбиение уменьшает неоднородность
и даёт большое значение $Q(S,j,t)$.
\end{frame}

\begin{frame}{Как дерево выбирает лучшее разбиение?}
\mlBody
\mlAlgorithm\; В каждой вершине рассматриваются допустимые
признаки $j$ и пороги $t$. Для каждой пары вычисляется $Q(S,j,t)$,
после чего выбирается
\[
 \boxed{(j^*,t^*)\in\operatorname*{arg\,max}_{j,t}Q(S,j,t).}
\]

\mlExample\; Рассмотрим две ситуации:
\begin{itemize}
  \item Если после разбиения каждая дочерняя вершина столь же
  неоднородна, как исходная, то $Q(S,j,t)=0$:
  качество не улучшилось.
  \item Если дочерние вершины стали однороднее, то $Q(S,j,t)>0$;
  более высокий $Q$ означает более удачный split.
\end{itemize}

\mlRemark\; Пустая дочерняя вершина недопустима. Для MSE,
индекса Джини и энтропии Шеннона взвешенная неоднородность
при разбиении не возрастает, поэтому $Q(S,j,t)\ge 0$.

\mlRemark\; Это \emph{жадное} решение: выбирается лучшее
разбиение текущей вершины, а не глобально оптимальное дерево.
\end{frame}

\begin{frame}{Числовой пример критерия Джини}
\mlBody
\mlExample\; В корне находятся четыре объекта: два класса $0$ и два класса $1$.
\[
 H_{\mathrm{Gini}}(S)=1-\left(\tfrac24\right)^2-\left(\tfrac24\right)^2=\tfrac12.
\]
Предположим, порог разделяет их на \emph{чистые} группы: слева оба объекта класса $0$, справа оба класса $1$.
\[
 H(S_L)=H(S_R)=0,\qquad Q(S,j,t)=\tfrac12-0=\boxed{\tfrac12}.
\]
Если же каждый дочерний узел снова содержит по одному объекту каждого класса, то
\[
 Q(S,j,t)=\tfrac12-\tfrac24\cdot\tfrac12-\tfrac24\cdot\tfrac12=0.
\]
    \mlConsequence\; Хороший split делает ответы дочерних вершин более однородными.
    Например, если в одном дочернем узле доли классов $3/4$ и $1/4$,
    то $H_{\mathrm{Gini}}=1-(3/4)^2-(1/4)^2=3/8<1/2$.

\end{frame}

\begin{frame}{Как выбирается порог среди кандидатов?}
\mlBody
Для фиксированного признака алгоритм сравнивает возможные пороги $t$ и выбирает тот, который сильнее уменьшает неоднородность $H(S)$.

\end{center}

\end{frame}

\begin{frame}{Как дерево строится: жадный алгоритм}
\mlBody
\mlAlgorithm
\begin{enumerate}
\item Положить $S=D_{\mathrm{train}}$ в корень.
\item В вершине перебрать допустимые признаки $j$ и пороги $t$.
\item Вычислить $Q(S,j,t)$, выбрать максимальное улучшение.
\item Разделить объекты на $S_L$ и $S_R$ и повторить рекурсивно.
\item Остановиться при малом числе объектов, предельной глубине или отсутствии улучшения.
\end{enumerate}
\mlRemark\; В листьях возвращается среднее $y$ для регрессии или наиболее частый класс для классификации.
\mlConsequence\; Глубина дерева контролирует сложность модели:
слишком глубокое дерево может подогнать и шум обучающей выборки.
\end{frame}

\begin{frame}{Структура обученного дерева}
\mlBody
Внутренняя вершина хранит условие $x_j\le t$, а лист --- прогноз.
Для нового объекта последовательно проверяем условия от корня до листа.

\end{center}
\mlRemark\; В отличие от линейной регрессии дерево задаёт
кусочно-постоянную функцию на областях пространства признаков.
\end{frame}



\begin{frame}{Bootstrap: как Random Forest получает разные деревья?}
\mlBody
\mlDefinition\; \textit{Bootstrap-выборка} --- новая выборка
размера $n$, полученная из $n$ исходных объектов случайными
выборами \emph{с возвращением}.

\mlExample\; Из объектов с номерами $1,2,3,4,5$ можно получить
индексы $(2,2,3,5,5)$: объекты $2$ и $5$ повторились,
а $1$ и $4$ не попали в данную выборку.

\mlAlgorithm\; В \textit{Random Forest (случайном лесе)}:
\begin{enumerate}
 \item Для каждого дерева независимо создают bootstrap-выборку.
 \item В каждой вершине случайно выбирают часть признаков и среди
 них ищут лучшее разбиение.
 \item Объединяют прогнозы деревьев (среднее или голосование).
\end{enumerate}

\mlRemark\; Эти две случайности уменьшают сходство деревьев.
Их усреднение помогает снизить разброс предсказаний,
но не гарантирует улучшения на любой выборке.
\end{frame}

\begin{frame}{Random Forest и Gradient Boosting: разная логика}
\mlBody
\mlStatement\; \textit{Random Forest} обучает деревья независимо
и объединяет их ответы. \textit{Gradient Boosting} обучает их
последовательно: каждое новое дерево дополняет уже построенную модель.

\[
 \text{Forest (регрессия): }F(x)=\frac1M\sum_{m=1}^{M}h_m(x),
 \qquad
 \text{Boosting: }F_m=F_{m-1}+\eta h_m.
\]
\mlRemark\; Для классификации лес использует голосование
или усреднение вероятностей; деревья бустинга зависят от предыдущих.
\end{frame}

\begin{frame}{Gradient Boosting: последовательное исправление ошибок}
\mlBody
\mlDefinition\; \textit{Градиентный бустинг} строит последовательность функций
\[
 \boxed{F_m(x)=F_{m-1}(x)+\eta h_m(x)},\qquad \eta>0,
\]
где $F_{m-1}$ --- ансамбль, $h_m$ --- новое дерево, $\eta$ --- шаг.

\mlRemark\; Здесь обновляется функция $F$, а не вектор параметров $\theta$.

\mlTask\; Пусть на выборке оптимизируется
\[
 L(F)=\frac1n\sum_{i=1}^n\ell\bigl(y_i,F(x_i)\bigr).
\]
Для дифференцируемой потери на шаге $m$ определяют \textit{псевдоостатки}
\[
 \boxed{r_i^{(m)}=-\left.\frac{\partial\ell(y_i,z)}{\partial z}\right|_{z=F_{m-1}(x_i)}}.
\]
Дерево $h_m$ обучается приближать значения $r_i^{(m)}$.
\end{frame}

\begin{frame}{При MSE бустинг приближает обычные остатки}
\mlBody
Возьмём квадратичную потерю (как в главе 2):
\[
 \ell(y,z)=(y-z)^2,\qquad
 -\frac{\partial\ell(y,z)}{\partial z}=2(y-z).
\]
Следовательно,
\[
 r_i^{(m)}=2\bigl(y_i-F_{m-1}(x_i)\bigr).
\]
Множитель $2$ можно учесть при подборе масштаба поправки. Содержательно дерево учится предсказывать остатки текущей модели.

\mlExample\; Если $y_i=10$, а $F_{m-1}(x_i)=7$, то остаток $3$; новое дерево должно предложить положительную поправку.

\mlConsequence\; Обычный GD меняет вектор $\theta$ по антиградиенту; Gradient Boosting строит новое направление в пространстве прогнозов посредством дерева $h_m$.
\end{frame}

\begin{frame}{Gradient Boosting: последовательность поправок}
\mlBody
\mlIdea\; Начальная функция $F_0$ постепенно улучшается добавлением
новых деревьев $h_m$:
\[
  F_m(x)=F_0(x)+\eta\sum_{k=1}^{m}h_k(x).
\]
\begin{center}
\slideimg[width=0.98\linewidth,height=4.5cm,keepaspectratio]{img/boosting_steps.pdf}
\end{center}
\mlRemark\; Каждая поправка обучается на псевдоостатках предыдущей модели.
\end{frame}

\begin{frame}{Gradient Boosting: как меняется ошибка}
\mlBody
Качество ансамбля на обучающей выборке оценивают той же функцией
эмпирического риска:
\[
 L(F_m)=\frac1n\sum_{i=1}^{n}\ell\bigl(y_i,F_m(x_i)\bigr).
\]
\begin{center}
\slideimg[width=0.91\linewidth,height=4.28cm,keepaspectratio]{img/boosting_loss.pdf}
\end{center}
\mlRemark\; График иллюстрирует один эксперимент.
Убывание ошибки на каждой итерации не гарантировано
для произвольного шага и структуры деревьев.
\end{frame}

\chapterframe{5}{Нейронные сети}

\begin{frame}{От линейной модели к искусственному нейрону}
\mlBody
В главе 2 предсказание имело вид $w^Tx+b$. \textit{Искусственный нейрон} добавляет к этой величине нелинейное преобразование:
\[
 \boxed{z=w^Tx+b,\qquad a=\varphi(z)}.
\]
\mlDefinition\; $w$ --- веса признаков, $b$ --- смещение (bias), $\varphi$ --- \textit{функция активации}.

\mlExample\; $x=(2,1)^T$, $w=(0.5,-1)^T$, $b=0.5$:
\[
 z=0.5\cdot2-1\cdot1+0.5=0.5.
\]
При $\varphi(z)=\max(0,z)$ получаем $a=0.5$.

\mlRemark\; Веса и смещения будут обучаемыми параметрами;
вектор всех параметров по-прежнему обозначается $\theta$.
По правилу цепочки для дифференцируемой $\varphi$:
\[
   \nabla_w a=\varphi'(z)x,\qquad \frac{\partial a}{\partial b}=\varphi'(z).
\]
Так простая модель уже требует производных для настройки своих весов.
\end{frame}

\begin{frame}{Почему нейрону нужна нелинейная активация?}
\mlBody
Композиция двух аффинных преобразований остаётся аффинной:
\[
 z_1=W_1x+b_1,\qquad z_2=W_2z_1+b_2,
\]
\[
 z_2=(W_2W_1)x+(W_2b_1+b_2).
\]
Без активаций несколько слоёв эквивалентны одному.

\mlDefinition\; Примеры активаций:
\[
 \operatorname{ReLU}(z)=\max(0,z),\qquad
 \sigma(z)=\frac1{1+e^{-z}},
\]
\[
 \tanh(z)=\frac{e^z-e^{-z}}{e^z+e^{-z}}.
\]
\mlConsequence\; Нелинейности позволяют сети выражать зависимости,
которые нельзя представить одной линейной моделью исходных признаков.
\mlRemark\; Для сигмоиды $\sigma'(z)=\sigma(z)(1-\sigma(z))$;
для ReLU производная равна $0$ при $z<0$ и $1$ при $z>0$
(в нуле обычная производная не существует).
\end{frame}

\begin{frame}{Многослойная сеть: последовательность функций}
\mlBody
\mlDefinition\; \textit{Multilayer Perceptron} (MLP) --- композиция слоёв, каждый из которых применяет линейное преобразование и нелинейность.

Обозначим вход $a^{(0)}=x$. Для слоя $\ell=1,\ldots,L$:
\[
 \boxed{z^{(\ell)}=W^{(\ell)}a^{(\ell-1)}+b^{(\ell)}},
\]
\[
 \boxed{a^{(\ell)}=\varphi_\ell(z^{(\ell)})}.
\]
Размеры: $W^{(\ell)}\in\mathbb R^{d_\ell\times d_{\ell-1}}$,
$b^{(\ell)}\in\mathbb R^{d_\ell}$.

\mlRemark\; Все $W^{(\ell)}$ и $b^{(\ell)}$ составляют один вектор
параметров $\theta$. Например, для двух слоёв:
\[
 f_\theta(x)=\varphi_2\!\left(W^{(2)}
 \varphi_1(W^{(1)}x+b^{(1)})+b^{(2)}\right).
\]
Выход $f_\theta(x)$ --- результат последнего слоя.
\end{frame}

\begin{frame}{Прямой проход: расчёт двухслойной сети}
\mlBody
\mlExample\; Возьмём $x=(1,2)^T$, два нейрона в скрытом слое
и один вероятностный выход. Пусть
\[
W^{(1)}=\begin{pmatrix}1&-1\\0&1\end{pmatrix},\quad
b^{(1)}=\begin{pmatrix}0\\-1\end{pmatrix},\quad
W^{(2)}=\begin{pmatrix}2&-1\end{pmatrix},\quad b^{(2)}=\tfrac12.
\]
\mlAlgorithm\; \textit{Forward pass} (прямое распространение):
\[
\begin{aligned}
 z^{(1)}&=W^{(1)}x+b^{(1)}=(-1,1)^T,\\
 a^{(1)}&=\operatorname{ReLU}(z^{(1)})=(0,1)^T,\\
 z^{(2)}&=W^{(2)}a^{(1)}+b^{(2)}=-\tfrac12,\\
 p&=\sigma(z^{(2)})\approx0.378.
\end{aligned}
\]
\mlRemark\; На каждом слое размерности матриц согласованы:
$W^{(1)}\in\mathbb R^{2\times2}$,
$W^{(2)}\in\mathbb R^{1\times2}$.
Если $y=0$, то BCE $=-\log(1-p)\approx0.474$.
\end{frame}

\begin{frame}{Пример архитектуры: распознавание цифр}

 \boxed{784\;\longrightarrow\;128\;\longrightarrow\;64\;\longrightarrow\;10}.
\]

\begin{frame}{Scores и Softmax: как сеть выдаёт вероятности?}
\mlBody
\mlDefinition\; \textit{Score} (также \textit{logit}) $z_k\in\mathbb R$
--- вещественная оценка класса $k$ \emph{до} перевода в вероятность.
Она может быть отрицательной и не обязана лежать между $0$ и $1$.

\mlDefinition\; \textit{Softmax} преобразует вектор
$z=(z_1,\ldots,z_K)^T$ в распределение по $K$ классам:
\[
 \boxed{p_k=\frac{e^{z_k}}{\sum_{j=1}^{K}e^{z_j}}},\qquad
 0<p_k<1,\quad\sum_{k=1}^{K}p_k=1.
\]

\mlExample\; Если $z=(0,\log 2,\log 3)^T$, то
\[
 (e^{z_1},e^{z_2},e^{z_3})=(1,2,3),\qquad
 p=\left(\tfrac16,\tfrac26,\tfrac36\right)^T.
\]

\mlRemark\; Чем больше score класса относительно остальных,
тем больше его вероятность. Прибавление одной константы
ко всем $z_k$ не изменяет результат Softmax.
\end{frame}

\begin{frame}{Выход сети и функция потерь}
\mlBody
Разные задачи требуют разных интерпретаций выхода сети
и разных функций потерь.

\mlDefinition\; \textit{Регрессия:} прогноз --- число
$\hat y=f_\theta(x)\in\mathbb R$. Например,
\[
 \boxed{\ell_{\mathrm{MSE}}(y,\hat y)=(y-\hat y)^2}.
\]

\mlDefinition\; \textit{Бинарная классификация:}
$p=\sigma(z)$ --- вероятность класса $1$,
$y\in\{0,1\}$. Используется
\[
 \boxed{\ell_{\mathrm{BCE}}(y,p)=
 -y\log p-(1-y)\log(1-p)}.
\]

\mlDefinition\; \textit{Многоклассовая классификация:}
$p=\operatorname{softmax}(z)$ --- распределение по $K$ классам.
При правильном классе $y\in\{1,\ldots,K\}$:
\[
 \boxed{\ell_{\mathrm{CE}}(y,p)=-\log p_y}.
\]

\mlRemark\; Общий принцип из главы 1 сохраняется:
прогноз $\to$ потеря $\to$ минимизация эмпирического риска.
\end{frame}

\begin{frame}{Softmax и Cross-Entropy: производная по score}
\mlBody
Для многоклассовой классификации $z=(z_1,\ldots,z_K)^T$ ---
выходные \textit{scores}, $p_k=e^{z_k}/\sum_j e^{z_j}$ --- вероятности.
Потеря при правильном классе $y$ равна
\[
  \ell(z,y)=-\log p_y=-z_y+\log\!\left(\sum_{j=1}^{K}e^{z_j}\right).
\]
\mlStatement\; Производная по каждой компоненте score:
\[
\begin{aligned}
 \frac{\partial\ell}{\partial z_k}
 &= -\mathbf 1\{k=y\}
 +\frac{e^{z_k}}{\sum_j e^{z_j}}\\
 &=\boxed{p_k-\mathbf 1\{k=y\}}.
\end{aligned}
\]
\mlExample\; При $p=(0.1,0.7,0.2)$ и правильном классе $y=2$:
\[
 \nabla_z\ell=(0.1,-0.3,0.2)^T.
\]
\mlConsequence\; Как и в логистической регрессии, производная имеет
форму «прогноз минус правильный ответ» (здесь последний закодирован
единицей в компоненте нужного класса).
\end{frame}

\begin{frame}{Обучение нейросети: тот же Gradient Descent}
\mlBody
Вектор $\theta$ содержит все веса и смещения, а $L(\theta)$
--- среднюю потерю на обучающей выборке:
\[
 \hat\theta\in\operatorname*{arg\,min}_{\theta}L(\theta).
\]
\mlAlgorithm\; Один шаг обучения включает три разные операции:
\begin{enumerate}
 \item \textit{Forward pass}: вычислить прогнозы $f_{\theta_t}(x_i)$.
 \item \textit{Backpropagation}: найти производные ошибки
 по всем весам и смещениям.
 \item \textit{Оптимизатор} (GD/Adam): использовать эти производные
 для обновления $\theta_{t+1}$.
\end{enumerate}
\mlStatement\; Например, при градиентном спуске
\[
 \boxed{\theta_{t+1}=\theta_t-\eta g_t},
\]
где $g_t$ --- градиент выбранной на шаге функции потерь.

\mlRemark\; Backpropagation \emph{вычисляет} градиент;
оптимизатор \emph{использует} его для изменения параметров.
\end{frame}

\begin{frame}{Mini-batch: градиент и его оценка}
\mlBody
\mlDefinition\; \textit{Mini-batch} $B_t$ --- небольшая случайная группа
объектов из обучающей выборки. Её функция потерь:
\[
 \boxed{L_{B_t}(\theta)=\frac1{|B_t|}\sum_{i\in B_t}
 \ell(y_i,f_\theta(x_i))}.
\]
\mlStatement\; Backpropagation вычисляет градиент именно этой функции:
\[
 \boxed{g_t=\nabla L_{B_t}(\theta_t)}.
\]
Поэтому для выбранного набора объектов градиент вычисляется
\emph{точно} (с учётом точности арифметики), но обычно
$g_t\ne\nabla L(\theta_t)$ для полной выборки.

\mlDefinition\; \textit{Эпоха (epoch)} --- один проход по всем
объектам обучающей выборки, обычно состоящий из многих mini-batch-шагов.

\mlConsequence\; Случайность появляется из выбора mini-batch,
а не из самого правила цепочки.
\end{frame}

\begin{frame}{Backpropagation: применение правила цепочки}
\mlBody
\mlDefinition\; \textit{Backpropagation} (обратное распространение ошибки) --- эффективное вычисление градиентов по весам сети с помощью правила цепочки.

\mlExample\; Для цепочки
\[
 h=\tanh(wx+b),\quad\hat y=vh+c,\quad\ell=(\hat y-y)^2
\]
вклад веса $w$ в ошибку составляет
\[
 \frac{\partial\ell}{\partial w}
 =\underbrace{2(\hat y-y)}_{\partial\ell/\partial\hat y}
 \cdot\underbrace{v}_{\partial\hat y/\partial h}
 \cdot\underbrace{(1-h^2)}_{\partial h/\partial(wx+b)}
 \cdot\underbrace{x}_{\partial(wx+b)/\partial w}.
\]
\mlConsequence\; Производные передаются от выхода к ранним слоям; промежуточные вычисления переиспользуются.

\mlRemark\; Backpropagation \emph{вычисляет} градиент, а оптимизатор \emph{обновляет} параметры по нему.
\end{frame}

\begin{frame}{Backpropagation: числовой пример}
\mlBody
\mlExample\; Рассмотрим простую сеть с одним скрытым нейроном:
\[
 h=\operatorname{ReLU}(wx+b),\quad
 \hat y=vh+c,\quad \ell=(\hat y-y)^2.
\]
Пусть $x=2$, $w=1$, $b=0$, $v=3$, $c=0$, $y=4$.
\[
 h=2,\qquad\hat y=6,\qquad\ell=(6-4)^2=4.
\]
\mlAlgorithm\; Двигаемся от потери к ранним параметрам:
\[
\begin{aligned}
 \frac{\partial\ell}{\partial\hat y}&=2(\hat y-y)=4,\\
 \frac{\partial\ell}{\partial v}&=4h=8,
 &\frac{\partial\ell}{\partial c}&=4,\\
 \frac{\partial\ell}{\partial h}&=4v=12,\\
 \frac{\partial\ell}{\partial w}&=12\cdot\operatorname{ReLU}'(2)\cdot x=24,
 &\frac{\partial\ell}{\partial b}&=12.
\end{aligned}
\]
\mlConsequence\; Каждая производная использует уже найденную
производную на следующем участке цепочки. Именно это делает
backpropagation вычислительно эффективным.
\end{frame}


\begin{frame}{Что в Backpropagation идёт «назад»?}
\mlBody
\mlIdea\; На прямом проходе сеть вычисляет выходы
слоёв $a^{(1)},\ldots,a^{(L)}$ и потерю $\ell$.
На обратном проходе выясняет, \emph{какие промежуточные значения
сильнее влияют на эту потерю}.

\mlDefinition\; \textit{Сигнал ошибки слоя} --- вектор
чувствительностей
\[
 \boxed{\delta^{(\ell)}=\nabla_{z^{(\ell)}}\ell},\qquad
 \delta_j^{(\ell)}=\frac{\partial\ell}{\partial z_j^{(\ell)}}.
\]

\mlStatement\; Для малого изменения выхода линейной части слоя
\[
 \Delta\ell\approx
 \bigl\langle\delta^{(\ell)},\Delta z^{(\ell)}\bigr\rangle.
\]
Положительная компонента $\delta_j^{(\ell)}$ означает, что
увеличение $z_j^{(\ell)}$ локально повышает потерю;
отрицательная --- понижает.

\mlRemark\; $\delta^{(\ell)}$ --- \emph{не} разность
«ответ минус прогноз», а производная потери по сигналам слоя.
Поэтому её можно вычислять последовательно от выхода ко входу.
\end{frame}

\begin{frame}{Матрица Якоби: правило цепочки для векторов}
\mlBody
\mlDefinition\; Для дифференцируемого отображения
$F:\mathbb R^d\to\mathbb R^m$ \textit{матрица Якоби}
содержит все частные производные его компонент:
\[
 \boxed{J_F(x)=\left(\frac{\partial F_i}{\partial x_j}\right)_
 {\substack{1\le i\le m\\1\le j\le d}}\in\mathbb R^{m\times d}}.
\]

\mlExample\; Если $F(x_1,x_2)=(x_1+x_2,x_1x_2)^T$, то
\[
 J_F(x)=\begin{pmatrix}1&1\\x_2&x_1\end{pmatrix}.
\]

\mlStatement\; Для скалярной функции $g:\mathbb R^m\to\mathbb R$
многомерное правило цепочки имеет вид
\[
 \boxed{\nabla_x(g\circ F)(x)
 =J_F(x)^T\nabla g\bigl(F(x)\bigr)}.
\]

\mlConsequence\; На обратном проходе производные
умножаются на транспонированный Якобиан очередного слоя.
Отсюда и возникает рекуррентная формула Backpropagation.
\end{frame}

\begin{frame}{Backpropagation: производные целого слоя}
\mlBody
Используем введённый сигнал ошибки слоя
$\delta^{(\ell)}=\nabla_{z^{(\ell)}}\ell$.
Символ $\odot$ обозначает покоординатное произведение векторов.

\mlStatement\; Если активации применяются покоординатно, то правило цепочки даёт
\[
 \boxed{\delta^{(\ell)}=
 \bigl(W^{(\ell+1)}\bigr)^T\delta^{(\ell+1)}
 \odot \varphi_\ell'\bigl(z^{(\ell)}\bigr)}.
\]
Из $z^{(\ell)}=W^{(\ell)}a^{(\ell-1)}+b^{(\ell)}$ следует
\[
 \boxed{\frac{\partial\ell}{\partial W^{(\ell)}}
 =\delta^{(\ell)}\bigl(a^{(\ell-1)}\bigr)^T},
 \qquad
 \frac{\partial\ell}{\partial b^{(\ell)}}=\delta^{(\ell)}.
\]
\mlRemark\; Так градиенты вычисляются от выхода сети ко входу;
их усредняют по mini-batch и передают оптимизатору из главы 3.
\end{frame}


\begin{frame}{Почему в глубоких сетях затухает градиент?}
\mlBody
\mlDefinition\; \textit{Затухание градиента} (vanishing gradient) --- ситуация, когда производные ошибки по параметрам ранних слоёв становятся очень малы.

\mlIdea\; При многократном применении правила цепочки возникает произведение производных последующих слоёв:
\[
 \frac{\partial\ell}{\partial h_1}
 =\frac{\partial\ell}{\partial h_L}
 \prod_{j=2}^{L}\frac{\partial h_j}{\partial h_{j-1}}
 \qquad\text{(для скалярной цепочки)}.
\]
Если множители по модулю меньше $1$, произведение может быстро уменьшаться.
Например, $0.5^{10}\approx0.001$ --- за десять слоёв масштаб градиента
уменьшился почти в тысячу раз.

\mlRemark\; Для векторных слоёв вместо произведения чисел
возникают произведения матриц Якоби, определённых ранее.
Не всякая глубокая сеть страдает от затухания градиента.
\end{frame}

\begin{frame}{ResNet: остаточные связи и путь для градиента}

\mlBody
\mlDefinition\; \textit{Остаточный блок} (residual block) вычисляет
\[
 \boxed{h_{\ell+1}=h_\ell+F_\ell(h_\ell)},
\]
где $F_\ell$ --- несколько обучаемых преобразований, а сложение требует согласованных размерностей.
\begin{center}
\begin{tikzpicture}[>=stealth,font=\small]
    \node (x) at (0,0) {$h_\ell$};
    \node[draw=defblue,thick,rounded corners=2pt,minimum width=2.45cm,minimum height=0.66cm] (f) at (3,0) {$F_\ell$ (несколько слоёв)};
    \node[draw=defblue,circle,inner sep=2pt] (sum) at (6.4,0) {$+$};
    \node (out) at (8.25,0) {$h_{\ell+1}$};
    \draw[->,gray!80] (x)--(f);
    \draw[->,gray!80] (f)--node[above,black] {$F_\ell(h_\ell)$}(sum);
    \draw[->,gray!80] (sum)--(out);
    \draw[->,defblue,thick] (x.north)--(0,1.0)--node[above,black] {обходная связь}(6.4,1.0)--(sum.north);
\end{tikzpicture}
\end{center}
\mlStatement\; В скалярном случае $dh_{\ell+1}/dh_\ell=1+F_\ell'(h_\ell)$;
для векторного входа якобиан блока равен
\[
 \boxed{J(h_\ell)=I+J_{F_\ell}(h_\ell)}.
\]
\mlRemark\; Добавка $I$ создаёт прямой путь для передачи производных через блок и часто облегчает обучение глубоких сетей. Она не гарантирует полного отсутствия затухания градиентов.
\end{frame}

\begin{frame}{CNN: локальные фильтры и общие веса}

\mlBody
\mlDefinition\; \textit{Convolutional Neural Network} (CNN, свёрточная сеть) обрабатывает локальные участки изображения обучаемыми фильтрами.
\begin{columns}[c,onlytextwidth]
    \column{0.41\textwidth}
    \centering
    \begin{tikzpicture}[x=0.40cm,y=0.40cm]
        \fill[blue!7] (0,0) rectangle (7,7);
        \foreach \x/\y in {2/5,3/5,4/5,4/4,3/3,2/2,2/1,3/1,4/1}
            {\fill[defblue!60] (\x,\y) rectangle +(1,1);}
        \draw[step=1,gray!55,line width=0.35pt] (0,0) grid (7,7);
        \draw[defblue,line width=1.5pt] (1,3) rectangle (4,6);
    \end{tikzpicture}
    \par\smallskip
    {\small Выделен участок $3\times3$.}
    \column{0.57\textwidth}
    \mlIdea\; Один фильтр последовательно перемещается по изображению и применяет \emph{те же самые веса} к разным участкам.
    \vspace{0.18cm}
    Это позволяет выделять локальные детали: штрихи, границы и более сложные сочетания.
\end{columns}
\vspace{0.12cm}
\mlStatement\; Для матрицы пикселей $A$ и фильтра $K\in\mathbb R^{r\times r}$ отклик в позиции $(u,v)$ равен
\[
 \boxed{B_{u,v}=\sum_{i=0}^{r-1}\sum_{j=0}^{r-1} K_{ij}A_{u+i,v+j}}.
\]
\mlRemark\; В ML это обычно кросс-корреляция (без переворота ядра). При шаге $1$ и отсутствии дополнения выход имеет размер $(H-r+1)\times(W-r+1)$.
\end{frame}

% Числовая иллюстрация свёрточного фильтра по chapter5.tex.
\begin{frame}{CNN: как фильтр обнаруживает границу}
\mlBody
\mlExample\; Возьмём фрагмент изображения размером $3\times3$ и фильтр, сравнивающий левый и правый столбцы:
\begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \centering
    \dblue{Участок изображения}
    \[
    A=\begin{pmatrix}1&1&0\\1&1&0\\1&1&0\end{pmatrix}
    \]
    \column{0.49\textwidth}
    \centering
    \dblue{Веса фильтра}
    \[
    K=\begin{pmatrix}1&0&-1\\1&0&-1\\1&0&-1\end{pmatrix}
    \]
\end{columns}
\mlAlgorithm\; Перемножим соответствующие элементы и сложим:
\[
 s=\sum_{i=1}^{3}\sum_{j=1}^{3}A_{ij}K_{ij}=1+1+1=3.
\]
\mlConsequence\; Если бы весь участок имел одинаковую яркость $c$, получилось бы $s=c\sum_{i,j}K_{ij}=0$. Ненулевой отклик отражает перепад яркости между его краями.
\mlRemark\; Перемещая этот фильтр по изображению, получаем \emph{карту признака}. В примере веса заданы вручную; в обучаемой CNN они подбираются по данным.
\end{frame}


\begin{frame}{CNN: почему общие веса экономят параметры?}
\mlBody
\mlDefinition\; \textit{Разделение весов} (weight sharing) означает,
что один фильтр $K$ применяется во всех положениях изображения.

\mlExample\; На входе изображение $28\times28$ с одним каналом.
Применим $8$ фильтров $3\times3$, каждый с одним смещением:
\[
 \boxed{N_{\mathrm{params}}=8(3\cdot3+1)=80}.
\]
Без дополнения и при шаге $1$ получим
$8$ карт размером $26\times26$, то есть $5408$ выходных значений.

Если бы у каждого положения был \emph{собственный} фильтр, пришлось бы
хранить $80\cdot26^2=54080$ параметров.

\mlConsequence\; Общие веса резко уменьшают число параметров
и позволяют применять один и тот же детектор признака
в разных частях изображения.

\mlRemark\; Для нескольких входных каналов формула числа параметров
меняется: один фильтр охватывает все входные каналы.
\end{frame}

\begin{frame}{Как CNN получает итоговый ответ}
\mlBody
\mlDefinition\; \textit{Карта признака} (feature map) --- значения отклика фильтра во всех доступных положениях. Несколько фильтров дают несколько карт.

Обычно вслед за фильтрами применяют ReLU, а размер карт иногда уменьшают с помощью \textit{pooling}.

\mlExample\; Max pooling по области $2\times2$:
\[
 \begin{pmatrix}1&3\\0&2\end{pmatrix}
 \longmapsto \max\{1,3,0,2\}=3.
\]
\mlConsequence\; Слои CNN постепенно формируют признаки от локальных штрихов к более сложным структурам, а последний классификационный слой (например, с softmax) выдаёт распределение по классам.

\mlRemark\; Фильтры и выходные слои обучаются совместно методом backpropagation и оптимизатором из главы 3.
\end{frame}


\begin{frame}{Нейросети: где здесь математический анализ?}
\mlBody
\mlStatement\; Нейронная сеть --- параметризованная композиция функций:
\[
 f_\theta=F_L\circ F_{L-1}\circ\cdots\circ F_1,
 \qquad \theta=\{W^{(\ell)},b^{(\ell)}\}_{\ell=1}^{L}.
\]
Цель обучения \emph{та же}, что в главе 1:
\[
 \hat\theta\in\operatorname*{arg\,min}_\theta
 \frac1n\sum_{i=1}^{n}\ell\bigl(y_i,f_\theta(x_i)\bigr).
\]
\mlAlgorithm\; Вычислительный цикл:
\begin{enumerate}
 \item Прямой проход --- последовательное вычисление композиции.
 \item Backpropagation --- правило цепочки для всех параметров.
 \item GD/Adam --- численное обновление параметров по градиенту.
\end{enumerate}
\mlConsequence\; В отличие от линейной регрессии сеть может строить
сложные нелинейные зависимости, но математически обучение снова
сводится к производным и оптимизации.
\end{frame}

\chapterframe{6}{Муравьиный алгоритм}


\begin{frame}{Задача коммивояжёра: точная постановка}
\mlBody
\mlTask\; Для $N$ городов найти \emph{кратчайший замкнутый маршрут},
который посещает каждый город ровно один раз до возвращения
в исходный город.

\mlDefinition\; Рассмотрим полный неориентированный взвешенный граф
$G=(V,E)$, где $V=\{1,\ldots,N\}$ --- города,
$d_{ij}=d_{ji}>0$ --- длина ребра между разными городами.

\mlDefinition\; Маршрут задаётся перестановкой
$\pi=(\pi_1,\ldots,\pi_N)$ вершин. Его длина:
\[
 \boxed{C(\pi)=\sum_{k=1}^{N-1}d_{\pi_k,\pi_{k+1}}
             +d_{\pi_N,\pi_1}}.
\]
Последнее слагаемое --- возвращение в начальный город.

\mlTask\; Найти $\pi^*\in\operatorname*{arg\,min}_{\pi}C(\pi)$.

\mlRemark\; Искомая переменная --- перестановка, а не вектор
вещественных параметров. Поэтому обычный градиентный спуск
по $\pi$ здесь неприменим; ACO ищет хороший маршрут вероятностно.
\end{frame}

\begin{frame}{Ant Colony Optimization: вероятностный выбор маршрута}
\mlBody
\mlTask\; Для задачи коммивояжёра найти допустимый тур $\pi$
с малой длиной $C(\pi)$, не перебирая все перестановки вершин.

\mlDefinition\; \textit{Ant Colony Optimization} (муравьиный алгоритм)
--- \emph{стохастическая метаэвристика}: алгоритм случайного поиска,
использующий правила выбора и подкрепления удачных вариантов.
Вероятность ребра зависит от феромона $\tau_{ij}>0$ и
привлекательности $\nu_{ij}>0$.
\[
 \boxed{P_{ij}=\frac{\tau_{ij}^{\alpha}\nu_{ij}^{\beta}}{\sum_{k\in A_i}\tau_{ik}^{\alpha}\nu_{ik}^{\beta}}},
 \qquad A_i\text{ --- ещё не посещённые города}.
\]
\mlExample\; Для расстояний $d_{ij}>0$ берут $\nu_{ij}=1/d_{ij}$.
Если два ребра имеют $\tau_1=2$, $\tau_2=1$,
$\nu_1=\nu_2=1$, $\alpha=\beta=1$, то
$P_1=\frac{2}{2+1}=\frac23$.

\mlRemark\; Параметры $\alpha,\beta\ge0$ управляют влиянием феромона и длины. В отличие от GD здесь оптимизируется дискретный маршрут, а не дифференцируемая функция параметров.
\end{frame}

\begin{frame}{Ant Colony Optimization: обновление феромона}
\mlBody
После построения маршрутов на итерации $t$ уровни феромона меняются:
\[
 \boxed{\tau_{ij}^{(t+1)}=(1-\rho_{\mathrm{evap}})\tau_{ij}^{(t)}+\Delta\tau_{ij}^{(t)}},
\]
где $0<\rho_{\mathrm{evap}}<1$ --- доля испарения, $\Delta\tau_{ij}^{(t)}$ --- добавка от найденных маршрутов.

\mlExample\; В простой версии для построенных маршрутов
$\pi_k$ длины $L_k$:
\[
 \Delta\tau_{ij}^{(t)}=
 \sum_{k:\,(i,j)\in\pi_k}\frac{\kappa}{L_k}.
\]
Для маршрутов длины $4$ и $8$ вклад каждого ребра первого
вдвое больше. Так короткие пути получают подкрепление.

\mlConsequence\; Случайный поиск сочетает исследование новых маршрутов и усиление уже найденных удачных решений.
\end{frame}

\begin{frame}{Муравьиный алгоритм: полный цикл поиска}
\mlBody
\mlTask\; Найти хороший маршрут среди большого числа дискретных
вариантов без перебора всех перестановок вершин.

\mlAlgorithm
\begin{enumerate}
 \item Задать начальные значения $\tau_{ij}>0$ на доступных рёбрах.
 \item Каждый «муравей» посещает все города без повторений,
 выбирая очередной по $P_{ij}$, и возвращается в стартовый город.
 \item Посчитать длину $L_k$ маршрута каждого муравья.
 \item Испарить часть феромона и подкрепить рёбра найденных путей:
 $\tau_{ij}\gets(1-\rho_{\mathrm{evap}})\tau_{ij}+\Delta\tau_{ij}$.
 \item Повторить построение маршрутов и запомнить лучший найденный.
\end{enumerate}

\mlRemark\; Чем короче маршрут, тем сильнее подкрепление
в рассмотренной версии алгоритма. Но случайность оставляет
ненулевой шанс исследования других допустимых продолжений.

\mlConsequence\; Это пример оптимизации \emph{без градиентов}:
ищем удачную комбинацию дискретных решений,
а не минимум дифференцируемой функции.
\end{frame}

\mlBody
В начале поиск распределён по многим рёбрам; после нескольких итераций
усиливаются пути, чаще встречающиеся в удачных маршрутах.

\begin{frame}{Итоги: единая математическая логика}
\mlBody
\begin{enumerate}
\item \textit{Постановка:} данные $D_{\mathrm{train}}$, модель $f_\theta$, потеря $\ell$ и задача $\min_\theta L(\theta)$.
\item \textit{Линейные модели:} MSE и BCE приводят к градиентам по параметрам.
\item \textit{Оптимизация:} GD и его модификации используют локальную информацию о $L(\theta)$.
\item \textit{Деревья и ансамбли:} дискретные разбиения и функциональная идея антиградиента в бустинге.
\item \textit{Нейросети:} композиции функций, правило цепочки и обратное распространение градиента.
\item \textit{Муравьиный алгоритм:} вероятностный поиск маршрута,
подкрепление удачных решений и испарение феромона без вычисления градиента.
\end{enumerate}
\mlConsequence\; Общая линия лекции --- постановка задачи,
выбор класса моделей, функция потерь и способ её уменьшения.
В нейросетях матанализ особенно нагляден: производные композиции
позволяют обучать сразу множество взаимосвязанных параметров.
Муравьиный алгоритм показывает альтернативу для дискретного поиска.
\end{frame}
